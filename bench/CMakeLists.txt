file(GLOB _lc_bench_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
add_executable(lanecast_bench ${_lc_bench_sources})
target_link_libraries(lanecast_bench PRIVATE lanecast)

# Head-to-head against libswscale / libavfilter linked directly. Optional:
# never required by the core build or CI. AUTO enables it when pkg-config
# finds libswscale 9 or newer (FFmpeg 8: SwsContext fields are public) and
# libavutil; libavfilter adds the filter and metric ops.
if(NOT LANECAST_BENCH_FFMPEG STREQUAL "OFF")
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(LANECAST_SWS QUIET IMPORTED_TARGET libswscale>=9 libavutil)
    pkg_check_modules(LANECAST_AVF QUIET IMPORTED_TARGET libavfilter)
  endif()
  if(LANECAST_SWS_FOUND)
    file(GLOB _lc_ff_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/ffmpeg/*.cpp")
    add_executable(bench_vs_ffmpeg ${_lc_ff_sources})
    target_include_directories(bench_vs_ffmpeg PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(bench_vs_ffmpeg PRIVATE __STDC_CONSTANT_MACROS)
    target_link_libraries(bench_vs_ffmpeg PRIVATE lanecast PkgConfig::LANECAST_SWS)
    if(LANECAST_AVF_FOUND)
      target_link_libraries(bench_vs_ffmpeg PRIVATE PkgConfig::LANECAST_AVF)
      target_compile_definitions(bench_vs_ffmpeg PRIVATE LANECAST_HAVE_AVFILTER=1)
    else()
      target_compile_definitions(bench_vs_ffmpeg PRIVATE LANECAST_HAVE_AVFILTER=0)
    endif()
    message(STATUS "lanecast: bench_vs_ffmpeg enabled (avfilter=${LANECAST_AVF_FOUND})")
  elseif(LANECAST_BENCH_FFMPEG STREQUAL "ON")
    message(FATAL_ERROR "LANECAST_BENCH_FFMPEG=ON but pkg-config cannot find libswscale and libavutil")
  else()
    message(STATUS "lanecast: bench_vs_ffmpeg disabled (libswscale not found)")
  endif()
endif()
